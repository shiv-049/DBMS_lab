create or replace procedure calcBonus
IS 
  Cursor emp_cursor IS
    select EmpID, Salary from Employee;
  Begin
    for emp_rec in emp_cursor loop

      update employee set bonus = emp_rec.Salary*0.10
      where EmpID = emp_rec.EmpID;

    END LOOP;

  END;
/

EXEC calcBonus;

create or replace function GetDeptCount(dept_name Varchar2)
return NUMBER
IS
  dept_count NUMBER;
  Begin
    select count(*) into dept_count from employee
    where Department = dept_name;

    return dept_count;
  END;
/

select GetDeptCount('IT') from dual;
