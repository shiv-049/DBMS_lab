##AND/OR
SELECT *
FROM Student
WHERE student_name LIKE 'A%'
AND email LIKE '%gmail%';

SELECT *
FROM Student
WHERE student_name LIKE 'A%'
OR student_name LIKE 'P%';

##MULTIPLICATION
SELECT course_id,
       course_name,
       credits,
       credits * 25 AS maximum_marks
FROM Course;

##Arithmetic + logical
SELECT course_name,
       credits,
       credits * 25 AS maximum_marks
FROM Course
WHERE credits >= 4;

#DIVISION
SELECT course_name,
       credits,
       100 / credits AS marks_per_credit
FROM Course;
